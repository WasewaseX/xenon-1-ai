   282→function stripFences(t) {

   287→function demoteHeadings(t) {
   288→  // The only allowed heading is line 1. Demote any later markdown headings to bold.
   289→  return t
   290→    .split("\n")
   291→    .map((line, i) => {
   292→      if (i === 0) return line;
   293→      const m = line.match(/^\s*#{1,6}\s+(.*)$/);
   294→      if (m) return m[1].trim() ? `**${m[1].trim()}**` : "";
   295→      return line;
   296→    })
   297→    .join("\n");
   298→}

   300→function splitRecap(raw) {
   301→  let body = raw, recap = "";
   302→  const lines = raw.split("\n");
   303→  for (let i = lines.length - 1; i >= 0; i--) {
   304→    const ln = lines[i];
   305→    if (/^\s*-{2,}\s*RECAP\s*-{2,}\s*$/i.test(ln) || /^\s*RECAP\s*:\s*/i.test(ln)) {
   306→      body = lines.slice(0, i).join("\n");
   307→      recap = lines.slice(i).join("\n").replace(/^\s*-{2,}\s*RECAP\s*-{2,}\s*/i, "").replace(/^\s*RECAP\s*:\s*/i, "").trim();
   308→      break;
   309→    }
   310→  }
   311→  return { body: body.trim(), recap: recap.trim() };
   312→}

   314→function cleanChapter(raw, ch) {
   315→  let t = stripFences(String(raw)).trim();
   316→  const { body, recap } = splitRecap(t);
   317→  t = body;
   318→  // drop trailing horizontal rules / whitespace
   319→  t = t.replace(/\n-{3,}\s*$/, "").trimEnd();
   320→  // drop accidental meta lines
   321→  t = t.split("\n").filter((ln) => !/^\s*\(?\s*word count\b/i.test(ln)).join("\n");
   322→  // collapse 3+ blank lines
   323→  t = t.replace(/\n{3,}/g, "\n\n").trim();
   324→  const expected = `# ${ch.label} — ${ch.title}`;
   325→  const lines = t.split("\n");
   326→  if (/^\s*#\s+/.test(lines[0])) {
   327→    lines[0] = expected;
   328→  } else {
   329→    lines.unshift(expected, "");
   330→  }
   331→  t = lines.join("\n");
   332→  t = demoteHeadings(t);
   333→  t = t.replace(/\n{3,}/g, "\n\n").trimEnd() + "\n";
   334→  return { text: t, recap };
   335→}
