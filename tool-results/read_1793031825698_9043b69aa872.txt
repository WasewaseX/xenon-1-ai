    66→    66→function dateGroupOf(iso: string): DateGroup {
    67→    67→  const d = new Date(iso);
    68→    68→  const now = new Date();
    69→    69→  const startOfDay = (x: Date) => new Date(x.getFullYear(), x.getMonth(), x.getDate()).getTime();
    70→    70→  const diffDays = Math.round((startOfDay(now) - startOfDay(d)) / 86_400_000);
    71→    71→  if (diffDays <= 0) return "today";
    72→    72→  if (diffDays === 1) return "yesterday";
    73→    73→  return "older";
    74→    74→}

   250→   250→
   251→   251→export function SidebarContent({ onNavigate }: { onNavigate?: () => void }) {
   252→   252→  const t = useT();
   253→   253→  const { toast } = useToast();
   254→   254→  const conversations = useXenon((s) => s.conversations);
   255→   255→  const folders = useXenon((s) => s.folders);
   256→   256→  const messages = useXenon((s) => s.messages);
   257→   257→  const activeId = useXenon((s) => s.activeId);
   258→   258→  const loadingSidebar = useXenon((s) => s.loadingSidebar);
   259→   259→  const searchQuery = useXenon((s) => s.searchQuery);
   260→   260→  const setSearchQuery = useXenon((s) => s.setSearchQuery);
   261→   261→  const newChat = useXenon((s) => s.newChat);
   262→   262→  const openConversation = useXenon((s) => s.openConversation);
   263→   263→  const renameConversation = useXenon((s) => s.renameConversation);
   264→   264→  const removeConversation = useXenon((s) => s.removeConversation);
   265→   265→  const createFolder = useXenon((s) => s.createFolder);
   266→   266→  const renameFolder = useXenon((s) => s.renameFolder);
   267→   267→  const removeFolder = useXenon((s) => s.removeFolder);
   268→   268→
   269→   269→  const [collapsedFolders, setCollapsedFolders] = useState<Set<string>>(new Set());
   270→   270→  const [renameTarget, setRenameTarget] = useState<ConversationDTO | null>(null);
   271→   271→  const [renameValue, setRenameValue] = useState("");
   272→   272→  const [deleteTarget, setDeleteTarget] = useState<ConversationDTO | null>(null);
   273→   273→  const [folderRenameTarget, setFolderRenameTarget] = useState<FolderDTO | null>(null);
   274→   274→  const [folderRenameValue, setFolderRenameValue] = useState("");
   275→   275→  const [folderDeleteTarget, setFolderDeleteTarget] = useState<FolderDTO | null>(null);
   276→   276→  const [newFolderOpen, setNewFolderOpen] = useState(false);
   277→   277→  const [newFolderName, setNewFolderName] = useState("");
   278→   278→  const [newFolderColor, setNewFolderColor] = useState(FOLDER_COLORS[0]);
   279→   279→
   280→   280→  const query = searchQuery.trim().toLowerCase();
   281→   281→  const matchesQuery = useMemo(
   282→   282→    () =>
   283→   283→      (c: ConversationDTO): boolean => {
   284→   284→        if (!query) return true;
   285→   285→        if (c.title.toLowerCase().includes(query)) return true;
   286→   286→        if (c.id === activeId && messages.some((m) => m.content.toLowerCase().includes(query))) return true;
   287→   287→        return false;
   288→   288→      },
   289→   289→    [query, activeId, messages]
   290→   290→  );
   291→   291→
   292→   292→  const pinned = useMemo(
   293→   293→    () => conversations.filter((c) => c.pinned && matchesQuery(c)).sort((a, b) => b.updatedAt.localeCompare(a.updatedAt)),
   294→   294→    [conversations, matchesQuery]
   295→   295→  );
   296→   296→  const ungrouped = useMemo(
   297→   297→    () => conversations.filter((c) => !c.folderId && !c.pinned && matchesQuery(c)).sort((a, b) => b.updatedAt.localeCompare(a.updatedAt)),
   298→   298→    [conversations, matchesQuery]
   299→   299→  );
   300→   300→
   301→   301→  const grouped = useMemo(() => {
   302→   302→    const groups: Record<DateGroup, ConversationDTO[]> = { today: [], yesterday: [], older: [] };
   303→   303→    for (const c of ungrouped) groups[dateGroupOf(c.updatedAt)].push(c);
   304→   304→    return groups;
   305→   305→  }, [ungrouped]);
   306→   306→
   307→   307→  const isEmpty = !loadingSidebar && conversations.length === 0;
   308→   308→  const totalMatches = useMemo(() => conversations.filter(matchesQuery).length, [conversations, matchesQuery]);
   309→   309→  const showNoResults = !loadingSidebar && query.length > 0 && totalMatches === 0;
   310→   310→
   311→   311→  const handleOpen = (id: string) => {
   312→   312→    openConversation(id).catch(() => undefined);
   313→   313→    onNavigate?.();
   314→   314→  };
   315→   315→
   316→   316→  const handleNewChat = () => {
   317→   317→    newChat();
   318→   318→    onNavigate?.();
   319→   319→  };
   320→   320→
   321→   321→  const handleCreateFolder = () => {
   322→   322→    const name = newFolderName.trim();
   323→   323→    if (!name) return;
   324→   324→    createFolder(name, newFolderColor)
   325→   325→      .then(() => {
   326→   326→        toast({ title: t("sidebar.toast.folderCreated") });
   327→   327→        setNewFolderOpen(false);
   328→   328→        setNewFolderName("");
   329→   329→        setNewFolderColor(FOLDER_COLORS[0]);
   330→   330→      })
   331→   331→      .catch(() => toast({ title: t("sidebar.toast.folderCreateFailed"), variant: "destructive" }));
   332→   332→  };
   333→   333→
   334→   334→  const confirmRenameConversation = () => {
   335→   335→    const title = renameValue.trim();
   336→   336→    if (!renameTarget || !title) return;
   337→   337→    renameConversation(renameTarget.id, title)
   338→   338→      .then(() => setRenameTarget(null))
   339→   339→      .catch(() => toast({ title: t("sidebar.toast.renameFailed"), variant: "destructive" }));
   340→   340→  };
   341→   341→
   342→   342→  const confirmRenameFolder = () => {
   343→   343→    const name = folderRenameValue.trim();
   344→   344→    if (!folderRenameTarget || !name) return;
   345→   345→    renameFolder(folderRenameTarget.id, name)
   346→   346→      .then(() => setFolderRenameTarget(null))
   347→   347→      .catch(() => toast({ title: t("sidebar.toast.folderRenameFailed"), variant: "destructive" }));
   348→   348→  };
   349→   349→
   350→   350→  const toggleFolder = (id: string) => {
   351→   351→    setCollapsedFolders((prev) => {
   352→   352→      const next = new Set(prev);
   353→   353→      if (next.has(id)) next.delete(id);
   354→   354→      else next.add(id);
   355→   355→      return next;
   356→   356→    });
   357→   357→  };
   358→   358→
   359→   359→  const renderRow = (c: ConversationDTO) => (
   360→   360→    <ConversationRow
   361→   361→      key={c.id}
   362→   362→      conversation={c}
   363→   363→      active={c.id === activeId}
   364→   364→      folders={folders}
   365→   365→      onOpen={() => handleOpen(c.id)}
   366→   366→      onRenameRequest={(conv) => {
   367→   367→        setRenameValue(conv.title);
   368→   368→        setRenameTarget(conv);
   369→   369→      }}
   370→   370→      onDeleteRequest={(conv) => setDeleteTarget(conv)}
   371→   371→    />
   372→   372→  );
   373→   373→
   374→   374→  return (
   375→   375→    <div className="flex h-full min-h-0 flex-col bg-sidebar">
   376→   376→      <div className="space-y-3 p-3">
   377→   377→        <Button className="h-11 w-full gap-2 text-sm font-semibold" onClick={handleNewChat}>
   378→   378→          <Plus className="h-4 w-4" aria-hidden />
   379→   379→          {t("sidebar.newChat")}
   380→   380→        </Button>
   381→   381→        <div className="relative">
   382→   382→          <Search className="pointer-events-none absolute start-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" aria-hidden />
   383→   383→          <Input
   384→   384→            id="xenon-sidebar-search"
   385→   385→            value={searchQuery}
   386→   386→            onChange={(e) => setSearchQuery(e.target.value)}
   387→   387→            placeholder={t("sidebar.searchPlaceholder")}
   388→   388→            aria-label={t("a11y.searchConversations")}
   389→   389→            className="h-10 ps-9"
   390→   390→          />
   391→   391→        </div>
   392→   392→      </div>
   393→   393→
   394→   394→      <nav aria-label={t("sidebar.conversations")} className="xenon-scroll min-h-0 flex-1 space-y-4 overflow-y-auto px-3 pb-4">
   395→   395→        {loadingSidebar ? (
   396→   396→          <div className="space-y-2" aria-label={t("a11y.loading")}>
   397→   397→            {[0, 1, 2, 3].map((i) => (
   398→   398→              <Skeleton key={i} className="h-10 w-full rounded-lg" />
   399→   399→            ))}
   400→   400→          </div>
   401→   401→        ) : isEmpty ? (
   402→   402→          <div className="flex flex-col items-center gap-2 px-4 py-12 text-center">
   403→   403→            <MessageSquare className="h-8 w-8 text-muted-foreground/50" aria-hidden />
   404→   404→            <p className="text-sm text-muted-foreground">{t("sidebar.emptyTitle")}</p>
   405→   405→            <p className="text-xs text-muted-foreground/70">{t("sidebar.emptyHint")}</p>
   406→   406→          </div>
   407→   407→        ) : showNoResults ? (
   408→   408→          <p className="px-4 py-10 text-center text-sm text-muted-foreground">{t("sidebar.noResults")}</p>
   409→   409→        ) : (
   410→   410→          <>
   411→   411→            {pinned.length > 0 ? (
   412→   412→              <section aria-label={t("sidebar.pinned")}>
   413→   413→                <h3 className="mb-1 flex items-center gap-1.5 px-2 text-[0.7rem] font-semibold uppercase tracking-wide text-muted-foreground">
   414→   414→                  <Pin className="h-3 w-3" aria-hidden />
   415→   415→                  {t("sidebar.pinned")}
   416→   416→                </h3>
   417→   417→                <ul className="space-y-0.5">{pinned.map(renderRow)}</ul>
   418→   418→              </section>
   419→   419→            ) : null}
   420→   420→
   421→   421→            <section aria-label={t("sidebar.folders")}>
   422→   422→              <h3 className="mb-1 flex items-center justify-between px-2 text-[0.7rem] font-semibold uppercase tracking-wide text-muted-foreground">
   423→   423→                <span className="flex items-center gap-1.5">
   424→   424→                  <FolderIcon className="h-3 w-3" aria-hidden />
   425→   425→                  {t("sidebar.folders")}
   426→   426→                </span>
   427→   427→                <button
   428→   428→                  type="button"
   429→   429→                  onClick={() => setNewFolderOpen(true)}
   430→   430→                  className="inline-flex h-6 items-center gap-1 rounded-md px-1.5 text-[0.7rem] font-normal normal-case text-muted-foreground transition-colors hover:bg-accent hover:text-foreground"
   431→   431→                  aria-label={t("a11y.newFolder")}
   432→   432→                >
   433→   433→                  <FolderPlus className="h-3.5 w-3.5" aria-hidden />
   434→   434→                  {t("sidebar.newFolder")}
   435→   435→                </button>
   436→   436→              </h3>
   437→   437→              {folders.length === 0 ? (
   438→   438→                <p className="px-2 py-1.5 text-xs text-muted-foreground/70">{t("sidebar.foldersHint")}</p>
   439→   439→              ) : (
   440→   440→                <ul className="space-y-0.5">
   441→   441→                  {folders.map((f) => {
   442→   442→                    const convs = conversations
   443→   443→                      .filter((c) => c.folderId === f.id && matchesQuery(c))
   444→   444→                      .sort((a, b) => b.updatedAt.localeCompare(a.updatedAt));
   445→   445→                    return (
   446→   446→                      <FolderRow
   447→   447→                        key={f.id}
   448→   448→                        folder={f}
   449→   449→                        folders={folders}
   450→   450→                        conversations={convs}
   451→   451→                        activeId={activeId}
   452→   452→                        expanded={!collapsedFolders.has(f.id)}
   453→   453→                        onToggle={() => toggleFolder(f.id)}
   454→   454→                        onOpenConversation={handleOpen}
   455→   455→                        onRenameRequest={(folder) => {
   456→   456→                          setFolderRenameValue(folder.name);
   457→   457→                          setFolderRenameTarget(folder);
   458→   458→                        }}
   459→   459→                        onDeleteRequest={(folder) => setFolderDeleteTarget(folder)}
   460→   460→                        onConversationRenameRequest={(conv) => {
   461→   461→                          setRenameValue(conv.title);
   462→   462→                          setRenameTarget(conv);
   463→   463→                        }}
   464→   464→                        onConversationDeleteRequest={(conv) => setDeleteTarget(conv)}
   465→   465→                      />
   466→   466→                    );
   467→   467→                  })}
   468→   468→                </ul>
   469→   469→              )}
   470→   470→            </section>
   471→   471→
   472→   472→            <section aria-label={t("sidebar.conversations")}>
   473→   473→              {ungrouped.length === 0 ? (
   474→   474→                <p className="px-2 py-1.5 text-xs text-muted-foreground/70">{t("sidebar.noUnfiled")}</p>
   475→   475→              ) : (
   476→   476→                (["today", "yesterday", "older"] as DateGroup[]).map((group) =>
   477→   477→                  grouped[group].length > 0 ? (
   478→   478→                    <div key={group} className="mb-3">
   479→   479→                      <h4 className="mb-1 px-2 text-[0.7rem] font-medium text-muted-foreground/80">{t(GROUP_LABEL_KEYS[group])}</h4>
   480→   480→                      <ul className="space-y-0.5">{grouped[group].map(renderRow)}</ul>
   481→   481→                    </div>
   482→   482→                  ) : null
   483→   483→                )
   484→   484→              )}
   485→   485→            </section>
   486→   486→          </>
   487→   487→        )}
   488→   488→      </nav>
   489→   489→
   490→   490→      {/* گفتگوی تغییر نام گفتگو */}
   491→   491→      <Dialog open={Boolean(renameTarget)} onOpenChange={(open) => !open && setRenameTarget(null)}>
   492→   492→        <DialogContent className="sm:max-w-sm">
   493→   493→          <DialogHeader>
   494→   494→            <DialogTitle>{t("sidebar.renameConversationTitle")}</DialogTitle>
   495→   495→            <DialogDescription>{t("sidebar.renameConversationDesc")}</DialogDescription>
   496→   496→          </DialogHeader>
   497→   497→          <Input
   498→   498→            value={renameValue}
   499→   499→            onChange={(e) => setRenameValue(e.target.value)}
   500→   500→            placeholder={t("sidebar.conversationTitlePlaceholder")}
   501→   501→            aria-label={t("a11y.conversationTitle")}
   502→   502→            onKeyDown={(e) => e.key === "Enter" && confirmRenameConversation()}
   503→   503→          />
   504→   504→          <DialogFooter>
   505→   505→            <Button variant="outline" onClick={() => setRenameTarget(null)}>
   506→   506→              {t("common.cancel")}
   507→   507→            </Button>
   508→   508→            <Button onClick={confirmRenameConversation} disabled={!renameValue.trim()}>
   509→   509→              {t("common.save")}
   510→   510→            </Button>
   511→   511→          </DialogFooter>
   512→   512→        </DialogContent>
   513→   513→      </Dialog>
   514→   514→
   515→   515→      {/* گفتگوی حذف گفتگو */}
   516→   516→      <AlertDialog open={Boolean(deleteTarget)} onOpenChange={(open) => !open && setDeleteTarget(null)}>
   517→   517→        <AlertDialogContent>
   518→   518→          <AlertDialogHeader>
   519→   519→            <AlertDialogTitle>{t("sidebar.deleteConversationTitle")}</AlertDialogTitle>
   520→   520→            <AlertDialogDescription>
   521→   521→              {t("sidebar.deleteConversationDesc", { title: deleteTarget?.title ?? "" })}
   522→   522→            </AlertDialogDescription>
   523→   523→          </AlertDialogHeader>
   524→   524→          <AlertDialogFooter>
   525→   525→            <AlertDialogCancel>{t("common.cancel")}</AlertDialogCancel>
   526→   526→            <AlertDialogAction
   527→   527→              className="bg-destructive text-white hover:bg-destructive/90"
   528→   528→              onClick={() => {
   529→   529→                if (!deleteTarget) return;
   530→   530→                removeConversation(deleteTarget.id).catch(() => undefined);
   531→   531→                setDeleteTarget(null);
   532→   532→              }}
   533→   533→            >
   534→   534→              {t("common.delete")}
   535→   535→            </AlertDialogAction>
   536→   536→          </AlertDialogFooter>
   537→   537→        </AlertDialogContent>
   538→   538→      </AlertDialog>
   539→   539→
   540→   540→      {/* گفتگوی تغییر نام پوشه */}
   541→   541→      <Dialog open={Boolean(folderRenameTarget)} onOpenChange={(open) => !open && setFolderRenameTarget(null)}>
   542→   542→        <DialogContent className="sm:max-w-sm">
   543→   543→          <DialogHeader>
   544→   544→            <DialogTitle>{t("sidebar.renameFolderTitle")}</DialogTitle>
   545→   545→            <DialogDescription>{t("sidebar.renameFolderDesc")}</DialogDescription>
   546→   546→          </DialogHeader>
   547→   547→          <Input
   548→   548→            value={folderRenameValue}
   549→   549→            onChange={(e) => setFolderRenameValue(e.target.value)}
   550→   550→            placeholder={t("a11y.folderName")}
   551→   551→            aria-label={t("a11y.folderNameInput")}
   552→   552→            onKeyDown={(e) => e.key === "Enter" && confirmRenameFolder()}
   553→   553→          />
   554→   554→          <DialogFooter>
   555→   555→            <Button variant="outline" onClick={() => setFolderRenameTarget(null)}>
   556→   556→              {t("common.cancel")}
   557→   557→            </Button>
   558→   558→            <Button onClick={confirmRenameFolder} disabled={!folderRenameValue.trim()}>
   559→   559→              {t("common.save")}
   560→   560→            </Button>
   561→   561→          </DialogFooter>
   562→   562→        </DialogContent>
   563→   563→      </Dialog>
   564→   564→
   565→   565→      {/* گفتگوی حذف پوشه */}
   566→   566→      <AlertDialog open={Boolean(folderDeleteTarget)} onOpenChange={(open) => !open && setFolderDeleteTarget(null)}>
   567→   567→        <AlertDialogContent>
   568→   568→          <AlertDialogHeader>
   569→   569→            <AlertDialogTitle>{t("sidebar.deleteFolder")}</AlertDialogTitle>
   570→   570→            <AlertDialogDescription>
   571→   571→              {t("sidebar.deleteFolderDesc", { name: folderDeleteTarget?.name ?? "" })}
   572→   572→            </AlertDialogDescription>
   573→   573→          </AlertDialogHeader>
   574→   574→          <AlertDialogFooter>
   575→   575→            <AlertDialogCancel>{t("common.cancel")}</AlertDialogCancel>
   576→   576→            <AlertDialogAction
   577→   577→              className="bg-destructive text-white hover:bg-destructive/90"
   578→   578→              onClick={() => {
   579→   579→                if (!folderDeleteTarget) return;
   580→   580→                removeFolder(folderDeleteTarget.id).catch(() => undefined);
   581→   581→                setFolderDeleteTarget(null);
   582→   582→              }}
   583→   583→            >
   584→   584→              {t("common.delete")}
   585→   585→            </AlertDialogAction>
   586→   586→          </AlertDialogFooter>
   587→   587→        </AlertDialogContent>
   588→   588→      </AlertDialog>
   589→   589→
   590→   590→      {/* گفتگوی ساخت پوشه */}
   591→   591→      <Dialog open={newFolderOpen} onOpenChange={setNewFolderOpen}>
   592→   592→        <DialogContent className="sm:max-w-sm">
   593→   593→          <DialogHeader>
   594→   594→            <DialogTitle>{t("sidebar.newFolderTitle")}</DialogTitle>
   595→   595→            <DialogDescription>{t("sidebar.newFolderDesc")}</DialogDescription>
   596→   596→          </DialogHeader>
   597→   597→          <Input
   598→   598→            value={newFolderName}
   599→   599→            onChange={(e) => setNewFolderName(e.target.value)}
   600→   600→            placeholder={t("a11y.folderName")}
   601→   601→            aria-label={t("a11y.folderNameInput")}
   602→   602→            onKeyDown={(e) => e.key === "Enter" && handleCreateFolder()}
   603→   603→          />
   604→   604→          <div className="flex items-center gap-2" role="radiogroup" aria-label={t("a11y.folderColor")}>
   605→   605→            {FOLDER_COLORS.map((color) => (
   606→   606→              <button
   607→   607→                key={color}
   608→   608→                type="button"
   609→   609→                role="radio"
   610→   610→                aria-checked={newFolderColor === color}
   611→   611→                aria-label={t("a11y.colorValue", { color })}
   612→   612→                onClick={() => setNewFolderColor(color)}
   613→   613→                className={cn(
   614→   614→                  "size-6 rounded-full border-2 transition-transform",
   615→   615→                  newFolderColor === color ? "scale-110 border-foreground" : "border-transparent hover:scale-105"
   616→   616→                )}
   617→   617→                style={{ backgroundColor: color }}
   618→   618→              />
   619→   619→            ))}
   620→   620→          </div>
   621→   621→          <DialogFooter>
   622→   622→            <Button variant="outline" onClick={() => setNewFolderOpen(false)}>
   623→   623→              {t("common.cancel")}
   624→   624→            </Button>
   625→   625→            <Button onClick={handleCreateFolder} disabled={!newFolderName.trim()}>
   626→   626→              {t("sidebar.createFolder")}
   627→   627→            </Button>
   628→   628→          </DialogFooter>
   629→   629→        </DialogContent>
   630→   630→      </Dialog>
   631→   631→    </div>
   632→   632→  );
   633→   633→}
